timeout 200
waitwin UI_MAIN_SCREEN 60000
wait 1000
click CP_MAIN_CTRL
waitwin CS_SELECT_WIN 20000
wait 2000
click SINGLE_COMMIT_CTRL

wait 2000

waitwin STARTCAMP_WIN 150000
wait 2000
click START_CAMP STARTCAMP_WIN
waitgone STARTCAMP_WIN 20000
wait 20000
hovericon CP_PUA_MAP 20 900
shot hover20
hovericon CP_PUA_MAP 60 900
shot hover60
hovericon CP_PUA_MAP 200 900
shot hover200
