# Enemy ships: unspotted (hidden) -> spotted (shown) -> lost again (dim ghost at last known spot).
timeout 200
waitwin UI_MAIN_SCREEN 60000
wait 1000
click CP_MAIN_CTRL
waitwin CS_SELECT_WIN 20000
wait 2000
click SINGLE_COMMIT_CTRL
waitwin STRAT_WIN 150000
wait 2000
click OK STRAT_WIN
waitgone STRAT_WIN 10000
wait 3000
listships CP_PUA_MAP
spotships 1
wait 5000
listships CP_PUA_MAP
shot spotted
wshot spotted_w
spotships 0
wait 5000
listships CP_PUA_MAP
shot lost
wshot lost_w
quit
